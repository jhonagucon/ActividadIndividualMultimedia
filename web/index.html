Colores de Clasificación

Césped — H:30°–90°, S≥40, Var>5
Tierra — H:5°–30°, S≥30, Var>15
Cemento — S<30, V≥100, Var≤15
Asfalto — S<35, V<100, Var>15
Otros — No clasificado
JavaScript · Canvas API 
// Para cada píxel (x, y) de la imagen:
const i = (y * W + x) * 4;
const {h, s, v} = rgbToHsv(data[i], data[i+1], data[i+2]);
const variance  = calcVarianza5x5(x, y, imageData);

// Clasificar por reglas HSV + varianza
if (h >= 30 && h <= 90 && s >= 40 && variance > 5)
  → CÉSPED  [0, 180, 0]
else if (h >= 5 && h < 30 && s >= 30 && variance > 15)
  → TIERRA  [160, 90, 42]
else if (s < 30 && v >= 100 && variance <= 15)
  → CEMENTO [200, 200, 200]
else if (s < 35 && v < 100 && variance > 15)
  → ASFALTO [72, 72, 72]
else
  → OTROS   [128, 0, 128]

JavaScript · Ventana deslizante 
for (let y = 0; y < H; y++) {
  for (let x = 0; x < W; x++) {
    let sumR=0, sumG=0, sumB=0, n=0;
    // Ventana 3×3 alrededor de (x, y)
    for (let ky=-1; ky<=1; ky++) {
      for (let kx=-1; kx<=1; kx++) {
        const idx = ((y+ky)*W+(x+kx))*4;
        sumR += src[idx];
        sumG += src[idx+1];
        sumB += src[idx+2];
        n++;
      }
    }
    // Promedio de los 9 vecinos
    out[i] = sumR/n;   // R
    out[i+1] = sumG/n; // G
    out[i+2] = sumB/n; // B
  }
}
